flowchart TD
  A["VALIDATE INTAKE<br/>[FUNCTION]<br/>in: rate-limited multipart form with 205 MB transport guard<br/>out: 1 to 50 images, ratios times variants at most 200 outputs, at most 10 refs and 200 MB files"]:::fn
  B["STORE FILES<br/>[LIBRARY · Supabase Storage]<br/>in: signature-checked files<br/>out: public trusted URLs"]:::data
  C["CREATE BATCH<br/>[DATA · Postgres transaction]<br/>in: settings and URLs<br/>out: batch and source times ratio times variant asset rows"]:::data
  D["ENQUEUE CALIBRATION<br/>[LIBRARY · BullMQ]<br/>in: batch ID<br/>out: retryable worker job"]:::data
  E["BUILD CREATIVE SPEC<br/>[AGENT · Gemini 3.5 Flash]<br/>in: brief, sources, image or short-video references<br/>out: normalized spec and weighted rubric"]:::agent
  F{"BATCH SPEND AVAILABLE?<br/>[FUNCTION + DATA]<br/>condition: atomic spend plus $0.018 to $0.045 input-aware estimate is within cap"}:::dec
  G["GENERATE CANDIDATE<br/>[LIBRARY · fal.ai FLUX.2 Klein 4B Base Edit]<br/>in: image 1 exact product, images 2 to 4 style only, chosen ratio, spec, 180 second timeout<br/>out: source-conditioned candidate image"]:::data
  H["UPSERT PROGRESS<br/>[DATA · Supabase and Postgres]<br/>in: successful candidate<br/>out: durable image without overwriting human decisions"]:::data
  I{"TEN CANDIDATES OR CAP?<br/>[FUNCTION]<br/>condition: 10 complete or spend unavailable"}:::dec
  J["APPROVE OR REJECT<br/>[ASK · human once]<br/>in: candidate<br/>out: decision and optional note"]:::ask
  K["SAVE DECISION<br/>[DATA · Postgres transaction]<br/>in: decision<br/>out: approval and candidate state"]:::data
  L{"APPROVED COUNT AT LEAST 5?<br/>[FUNCTION]<br/>condition: durable approvals at least 5"}:::dec
  M["SYNTHESIZE RECIPE<br/>[AGENT · Gemini 3.5 Flash]<br/>in: positives, negatives, notes, rubric<br/>out: 2 to 5 ranked prompts and tightened rubric"]:::agent
  N["READY FOR AUTONOMY<br/>[DATA · Postgres]<br/>in: recipe<br/>out: ready batch"]:::term
  O["CALIBRATION ERROR<br/>[DATA · visible worker failure]<br/>in: fewer than 5 durable candidates<br/>out: retry or DLQ evidence"]:::term

  I -->|"Need more"| F
  I -->|"Enough to review"| J --> K --> L
  L -->|"No"| J
  L -->|"Yes"| M --> N
  F -->|"No and fewer than 5"| O

%% 03 Autonomous per-asset flow
%% files: src/lib/orchestrator/graph.ts, src/lib/providers/fal.ts, src/lib/providers/gemini.ts, src/lib/repository.ts, worker/index.ts
%% granularity: full
flowchart TD
  A["CONSUME ASSET JOB<br/>[LIBRARY · BullMQ worker]<br/>in: batch ID and asset ID<br/>out: hydrated durable state"]:::data
  B{"OUTPUT ALREADY AWAITING QC?<br/>[FUNCTION]<br/>condition: status is qc and output URL exists"}:::dec
  C["RESUME QC WITHOUT REGENERATION<br/>[FUNCTION]<br/>in: stored output<br/>out: QC-ready state"]:::fn
  D{"ATTEMPT AND ASSET CAP OK?<br/>[FUNCTION]<br/>condition: attempts below 5 and input-aware $0.018 to $0.045 estimate at most cap"}:::dec
  E{"BATCH CAP RESERVATION SUCCEEDS?<br/>[DATA · atomic Postgres update]<br/>condition: batch spend plus estimate at most persisted cap"}:::dec
  F["SAVE GENERATING ATTEMPT<br/>[DATA · Postgres transaction]<br/>in: prompt, params, estimated cost<br/>out: durable attempt boundary"]:::data
  G["GENERATE OR REVISE<br/>[LIBRARY · fal.ai FLUX.2 Klein 4B Base Edit]<br/>in: image 1 exact product, images 2 to 4 style only, requested ratio, prompt, QC correction, bounded timeout<br/>out: one ratio-correct source-conditioned image"]:::data
  H["STORE OUTPUT<br/>[LIBRARY · Supabase Storage]<br/>in: trusted provider image at most 25 MB<br/>out: durable output URL"]:::data
  I["RUBRIC JUDGMENT<br/>[AGENT · Gemini 3.5 Flash]<br/>in: source, output, approved refs, spec, bounded timeout<br/>out: five scores, feedback, corrections"]:::agent
  J["REFERENCE SIMILARITY<br/>[LIBRARY · Gemini Embedding 2]<br/>in: output and up to five approvals<br/>out: maximum cosine similarity"]:::data
  K{"PASS POLICY<br/>[FUNCTION]<br/>condition: blended score at least configured threshold and fidelity at least 85"}:::dec
  L["ACCEPT AND RECORD WIN<br/>[DATA · Postgres]<br/>in: passing QC and prompt ID<br/>out: passed asset and updated win rate"]:::term
  M{"FEEDBACK RETRIES LEFT?<br/>[FUNCTION]<br/>condition: retries below configured K, default 2"}:::dec
  N{"NEXT RANKED PROMPT EXISTS?<br/>[FUNCTION]<br/>condition: another recipe and attempts remain"}:::dec
  O["PREPARE CORRECTION<br/>[FUNCTION]<br/>in: QC corrections<br/>out: feedback-guided next state"]:::fn
  P["NEEDS REVIEW<br/>[DATA · Postgres]<br/>in: cap or exhausted strategy<br/>out: terminal asset and visible reason"]:::term
  Q{"ALL SIBLINGS TERMINAL?<br/>[FUNCTION]<br/>condition: no queued, generating, or qc assets"}:::dec
  R["BATCH DONE AND DELIVERABLE<br/>[DATA · Postgres]<br/>in: final terminal asset, then successful export<br/>out: completed timestamp and delivered states"]:::term

  M -->|"Yes"| O --> D
